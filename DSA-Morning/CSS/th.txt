syntax of css

    selector : id/ class/ tag

    property : color, font-size, background-color, margin, padding, border, width, height

    value : red, 20px, #fff, 10px, 5px solid black, 100%, auto


    selector{
        property : value;
        property : value;
        property : value;
        property : value;
    }



    h1{
        color : red;
    }


    for id call we need to use #(hash) id_name

    for class call we need to use .(dot) class_name
